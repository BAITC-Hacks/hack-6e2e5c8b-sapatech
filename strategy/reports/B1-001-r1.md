```python
"""B1-001: audit known final IDs and estimated marginal gain, no pilot IDs."""
import json
import sys
from time import perf_counter
from types import SimpleNamespace
from unittest.mock import patch

import numpy as np
import pandas as pd

sys.path.append('/private/tmp/hackalem-t04-case.MYwV6x')
from mock_environment import make_mock_env
from agent import Agent
import strategy
from strategy import core


def audit(seed):
    captured = {}
    def select(*args):
        captured['args'] = args
        return strategy.select_campaigns(*args)
    policy = SimpleNamespace(build_candidates=strategy.build_candidates,
                             choose_pilot=strategy.choose_pilot, select_campaigns=select)
    env, _ = make_mock_env(seed=seed)
    agent = Agent(strategy=policy)
    started = perf_counter()
    campaigns = agent.act(env)
    elapsed = perf_counter() - started
    assert agent.diagnostics['errors'] == []
    raw_profile, tariffs, raw_channels, candidates, observations, resources = captured['args']
    profile = core._profile(raw_profile, tariffs)
    channels = core._channels(raw_channels)
    observed = core._observations(observations)
    positions = core._candidate_positions(profile, candidates)
    arpu = profile.predicted_arpu.to_numpy(dtype=float)
    p, pilot_best = np.zeros(len(profile)), np.zeros(len(profile))
    for c in candidates:
        ids = positions[c['candidate_id']]
        for channel, n, ratio in observed.get(c['candidate_id'], []):
            mean, _ = core._estimate(c, channel, channels, observed)
            p[ids] = 1 - (1 - p[ids]) * (1 - min(n / len(ids), 1))
            pilot_best[ids] = np.maximum(pilot_best[ids], mean * arpu[ids])
    # Independently evaluate the whole prefix objective before/after each action.
    best = np.full(len(profile), -np.inf)
    objective = np.zeros(len(profile))
    budget, contacts = resources['remaining_budget'], resources['remaining_contacts']
    rows = []
    for campaign in campaigns:
        filters = {k: campaign[k] for k in core.FILTERS if k in campaign}
        c = next(c for c in candidates if c['filters'] == filters and c['target_tariff'] == campaign['target_tariff'])
        channel = campaign['channel']
        cost = channels[channel][0]
        n = min(resources['max_customers_per_campaign'], contacts)
        if cost:
            n = min(n, int(budget // cost))
        ids = positions[c['candidate_id']][:n]
        mean, error = core._estimate(c, channel, channels, observed)
        lift = (mean - 0.5 * error) * arpu[ids]
        repeated = np.isfinite(best[ids])
        next_best = np.maximum(best[ids], lift)
        next_objective = (1 - p[ids]) * next_best + p[ids] * (np.maximum(next_best, pilot_best[ids]) - pilot_best[ids])
        delta = next_objective - objective[ids]
        net = delta - cost
        assert np.isfinite(net).all()
        assert not rows or float(net.sum()) > 0
        row = dict(name=campaign['campaign_name'], channel=channel,
                   target=campaign['target_tariff'], filters=filters,
                   contacts=len(ids), repeated=int(repeated.sum()),
                   no_lift_improvement=int((repeated & (lift <= best[ids])).sum()),
                   cost=float(len(ids) * cost), repeat_cost=float(repeated.sum() * cost),
                   estimated_new_net=float(net[~repeated].sum()),
                   estimated_repeat_net=float(net[repeated].sum()),
                   estimated_incremental_net=float(net.sum()))
        contacts -= len(ids)
        budget -= len(ids) * cost
        row.update(contacts_left=contacts, budget_left=budget)
        rows.append(row)
        objective[ids] = next_objective
        best[ids] = next_best
    assert contacts >= 0 and budget >= 0
    result = dict(seed=seed, source='public profile and pilot observations only',
                  runtime_seconds=elapsed, pilots=len(observations),
                  pilot_contacts=sum(o['result']['n_customers'] for o in observations),
                  final_contacts=sum(r['contacts'] for r in rows),
                  final_repeats=sum(r['repeated'] for r in rows),
                  repeated_without_lift_improvement=sum(r['no_lift_improvement'] for r in rows),
                  repeat_cost=sum(r['repeat_cost'] for r in rows),
                  final_cost=sum(r['cost'] for r in rows),
                  estimated_final_incremental_net=sum(r['estimated_incremental_net'] for r in rows),
                  final_contacts_left=contacts, final_budget_left=budget,
                  violations=0, campaigns=rows)
    print(json.dumps(result, ensure_ascii=False), flush=True)


def controlled_checks():
    profile = pd.DataFrame({'ID_NUMBER': list(range(100)), 'current_tariff': ['tariff_1'] * 100,
                            'arpu_segment': ['HIGH'] * 100, 'predicted_arpu': [1000.] * 100})
    tariffs = pd.DataFrame({'tariff_plan_code': ['tariff_1', 'tariff_2', 'tariff_3']})
    candidates = strategy.build_candidates(profile, tariffs, pd.DataFrame())
    resources = dict(remaining_budget=10000, remaining_contacts=500, max_campaigns=10,
                     max_customers_per_campaign=5000, pilots_left=0, seconds_left=200)
    channels = {'sms': {'cost_per_contact': 4, 'conversion_multiplier': .65},
                'call': {'cost_per_contact': 160, 'conversion_multiplier': 1.2}}
    # With identical effects there is no incremental gain, and all repeat costs remain.
    with patch.object(core, '_estimate', return_value=(.2, 0)):
        equal = strategy.select_campaigns(profile, tariffs, channels, candidates, [], resources)
    assert len(equal) == 1 and equal[0]['channel'] == 'sms'
    # Call prefix has +6200 vs SMS +5600, then only 80 budget remains.
    # SMS can reach only the same first 20 IDs and must be rejected.
    def estimate(candidate, channel, *unused):
        return (.26 if channel == 'call' else .06), 0
    with patch.object(core, '_estimate', side_effect=estimate):
        mixed = strategy.select_campaigns(profile, tariffs, channels, candidates, [], resources)
    assert [c['channel'] for c in mixed] == ['call']  # only 80 money left: no new IDs reachable
    resources['remaining_budget'] = 10400
    # Check an exact-alternative case with per-campaign cap; no repeated effect multiplication.
    channels = {'push': {'cost_per_contact': 0, 'conversion_multiplier': .5}}
    with patch.object(core, '_estimate', return_value=(.2, 0)):
        duplicate = strategy.select_campaigns(profile, tariffs, channels, candidates, [], resources)
    assert len(duplicate) == 1
    print(json.dumps({'controlled_checks': 'passed', 'paid_equal_effect_campaigns': len(equal),
                      'insufficient_budget_prefix_campaigns': len(mixed),
                      'free_equal_effect_campaigns': len(duplicate)}))


if __name__ == '__main__':
    controlled_checks()
    for seed in (0, 42):
        audit(seed)
```
